เทสต์นี้รันผ่านกับ connector `hr` ใน[ตัวอย่าง path parameter และ POST](/docs/connect/examples/rest-path-post) ใช้ `bun test`

```ts
import { afterAll, beforeAll, describe, expect, test } from "bun:test";
import type { AccessContext } from "@/lib/contracts";
import { USERS } from "@/lib/data/entities/users";
import { liveAccessFor } from "@/lib/access/enforce";
import { runWithAccess } from "@/lib/server/request-context";
import { winyuTool } from "@/lib/server/tools/registry";
import { registerConnectors, resetConnectors } from "@/lib/server/connectors";
import { registerRestTransport, resetRestTransport } from "@/lib/server/connectors/rest";
import { hrConnector } from "./hr";

const SALARY_ROWS = [
  { employee_id: "e_1", region: "northeast", salary: 52000, national_id: "1101700000001" },
  { employee_id: "e_2", region: "bkk", salary: 61000, national_id: "1101700000002" },
];

const sent: Request[] = [];

beforeAll(() => {
  registerConnectors([hrConnector]);
  registerRestTransport(async (request) => {
    sent.push(request);
    return Response.json({ rows: SALARY_ROWS });
  });
});

afterAll(() => {
  resetConnectors();
  resetRestTransport();
});

function accessOf(userId: string): AccessContext {
  const user = USERS.find((item) => item.id === userId);
  if (!user) throw new Error(`no user ${userId}`);
  return liveAccessFor(user);
}

async function ask(userId: string, tool: string, input: Record<string, unknown>) {
  const execute = winyuTool(tool)?.tool.execute as (input: unknown, options: unknown) => Promise<Record<string, unknown>>;
  return runWithAccess(accessOf(userId), () => execute(input, {}));
}

describe("hr__salary_history", () => {
  test("a sales rep cannot call it at all", async () => {
    const result = await ask("u_krit", "hr__salary_history", { department: "ขาย", year: 2026, regions: null });
    expect(result.code).toBe("TOOL_NOT_ALLOWED");
    expect(sent).toHaveLength(0);
  });

  test("a regional manager gets only their region, salary masked, national id gone", async () => {
    const result = (await ask("u_anucha", "hr__salary_history", { department: "ขาย", year: 2026, regions: "all" })) as { rows: Record<string, unknown>[] };
    expect((await sent.at(-1)?.json()).regions).toBe("northeast");
    expect(result.rows).toEqual([{ employee_id: "e_1", region: "northeast", salary: "***" }]);
  });
});
```

## เทสต์นี้พิสูจน์อะไร

- **บทบาท** พนักงานขายเรียก tool ไม่ได้ และไม่มีคำขอหลุดไปถึงระบบต้นทาง (`sent` ว่าง)
- **`inject`** AI ส่ง `regions: "all"` แต่คำขอที่ไปถึงระบบต้นทางคือ `northeast`
- **`filter`** ระบบจำลองตอบกลับทุกภาค (เหมือนระบบที่ไม่สนใจตัวกรอง) แต่ผู้จัดการภาคอีสานได้แถวเดียว
- **`sensitive`** `salary` เป็น `***` และ `national_id` หายไปทั้งคีย์

## เทสต์ที่ควรมีเพิ่ม

ดูชุดเต็มของ CRM เดโมได้ที่ `lib/access/redteam-rest.test.ts` ซึ่งครอบคลุม

- คำขอมีตัวตนที่ลงลายเซ็นแล้ว ไม่ตาม redirect และระบบจำลองปฏิเสธคำขอที่ลายเซ็นไม่ถูก
- ขอเอเย่นต์ของภาคอื่น ได้คำตอบว่าอยู่นอกขอบเขต และ audit บันทึกเป็น `deny`
- ข้อความจากระบบต้นทางที่มีคำสั่งแฝงถูกกั้นไว้เป็นข้อมูลก่อนถึง AI
- ระบบต้นทางไม่ตอบหรือตอบ 502–504 ได้ `CONNECTOR_UNAVAILABLE` และ connector ขึ้นว่าไม่ตอบในหน้า admin
- ระบบต้นทางปฏิเสธ หรือ body ไม่ตรงกับ adapter ได้ `CONNECTOR_FAILED`
- admin ปิด connector หรือปิด tool ของบทบาทหนึ่ง แล้วคำขอไม่ไปถึงระบบต้นทาง

วิธีตรวจว่าเทสต์ใช้ได้จริง: ถอด `filter` ออกจาก config ชั่วคราวแล้วรันใหม่ เทสต์เรื่องข้อมูลรั่วต้องล้ม ถ้าไม่ล้ม แปลว่าเทสต์ไม่ได้ตรวจอะไร
